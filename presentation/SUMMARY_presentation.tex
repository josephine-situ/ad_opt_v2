\begin{frame}{Data}
  \textbf{Two sources}
  \begin{itemize}
    \item \texttt{campaign-summary} --- Google Ads \textbf{change history} (marketing decisions): 47 campaign versions, 25 keyword sets, \textbf{4} campaign-level match configurations (\texttt{Broad}, \texttt{Exact}, \texttt{Phrase+Exact}, \texttt{Broad+Phrase+Exact}).
    \[
      (\text{daily\_budget},\ \text{region},\ \text{match\_types},\ \text{keyword\_set\_id}).
    \]
    \item \texttt{kw-day-panel} --- \textbf{Search keyword} reports (performance data): clicks, cost, all conv.\ (36k keyword-day observations, last 2 years); \textbf{3} keyword-level match types (\texttt{Broad}, \texttt{Phrase}, \texttt{Exact}).
  \end{itemize}
  \vspace{0.5em}
  \textbf{Modeling panel:} \texttt{campaign-day-panel} --- merge yields 2{,}222 campaign-days (12 region $\times$ match-config segments; \texttt{match\_types} from summary), each linked to a \texttt{campaign\_version}.
\end{frame}

\begin{frame}{Initial data analysis}
  \begin{itemize}
    \item \textbf{Budget vs.\ clicks:} positive relationship, but slopes differ --- B strong at low budget; USA needs high spend for modest gains.
    \item \textbf{Temporal stability:} click rate (clicks/budget) shifts at keyword-set boundaries --- train on recent windows or add set fixed effects.
    \item \textbf{Keyword sets:} high turnover (median 43\% keywords retained); model \textbf{set-level} semantic features (cohesion, course similarity) rather than individual keywords.
    \item \textbf{Keyword count vs.\ clicks:} $\text{correlation} \approx 0.05$--$0.08$ --- list size is not the driver.
  \end{itemize}
\end{frame}

\begin{frame}{Models}
  \footnotesize
  \textbf{Per-segment:} $\text{clicks} = a\,\text{budget}^{b}$;\;
  marginal $= a b\,\text{budget}^{b-1}$ \;(5/12 segments, $\geq 3$ budget levels).\\
  \textbf{Pooled (M1--M3):} $\log y = \alpha_{\text{region}} + \mu_{\text{match}} + b\log(\text{budget}) + \varepsilon$.\\
  \textbf{Linear (M4):} $y = \alpha_s + \beta_s\,\text{budget} + \mathbf{z}_t^\top\boldsymbol{\gamma} + \varepsilon$
  \;(M4b: per-segment $\beta_s$ + calendar + keyword embeddings).

  \vspace{0.4em}
  \centering

\end{frame}

\begin{frame}{Main findings}
  \begin{itemize}
    \item \textbf{B / Phrase; Exact} lead linear M4b click slopes ($\beta_s \approx 0.7$--2.4 clicks/\$1); conv.\ still favors B / Phrase; Exact ($\approx$ 0.14 conv/\$1 vs.\ USA Broad $\approx$ 0.02).
    \item \textbf{Clicks $\neq$ conversions} in USA and Broad --- USA: click elasticity (\% change in clicks / \% change in budget) $\approx 0.82$, conversion elasticity (\% change in conversions / \% change in budget)$\approx 0.21$; Broad segments $r < 0.1$ between clicks and conv.
    \item \textbf{Sparse segments:} 7/12 have one historical budget $\Rightarrow$ fit one model rather than per-segment models.
  \end{itemize}
\end{frame}

\begin{frame}{Optimization formulation}
  \small
  \textbf{Segments} $s \in \mathcal{S}$: region $\times$ match type.
  \textbf{Candidates} $\mathcal{K}_s$: feasible \texttt{keyword\_set\_id} for $s$
  (historical sets in segment + lists validated on \texttt{kw-day-panel}).

  \textbf{Decision variables:}
  $x_s \geq 0$ daily budget;\;
  $y_{s,k} \in \{0,1\}$ with $\sum_{k \in \mathcal{K}_s} y_{s,k} = 1$ (one keyword set per segment).

  \textbf{Response (M4b, all conv.; planning-day calendar $\mathbf{z}^{\mathrm{cal}}$ fixed):}
  \[
    \widehat{\mathrm{conv}}_{s}
      = \alpha_s + \beta_s\, x_s
      + \sum_{k \in \mathcal{K}_s} y_{s,k}\,\boldsymbol{\phi}_k^\top \boldsymbol{\gamma}_{\mathrm{emb}}
      + {\mathbf{z}^{\mathrm{cal}}}^{\!\top} \boldsymbol{\gamma}_{\mathrm{cal}}
  \]
  $\boldsymbol{\phi}_k$: embedding features for set $k$;
  $\beta_s$ shrunk toward pooled estimate when segment has sparse budget history.

  \textbf{Problem (MILP):}
  \[
    \max_{\{x_s,\,y_{s,k}\}} \sum_{s \in \mathcal{S}} \widehat{\mathrm{conv}}_{s}
    \quad \text{s.t.}\quad
    \sum_{s} x_s \leq B,\;
    x^{\min}_s \leq x_s \leq x^{\max}_s,
  \]
  \[
    \sum_{s:\, r(s)=\mathrm{USA}} \!\!x_s
    \;\geq\;
    \sum_{s:\, r(s)=A} \!\!x_s
    \;\geq\;
    \sum_{s:\, r(s)=B} \!\!x_s,
    \qquad
    \sum_{k \in \mathcal{K}_s} y_{s,k} = 1 \;\;\forall s.
  \]
  \normalsize
\end{frame}
