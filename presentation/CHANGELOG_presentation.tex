\begin{frame}{Scope}

  \vspace{0.75em}

    \item Keyword set pipeline

  \end{enumerate}
\end{frame}

\begin{frame}{MILP \& optimizer correctness}
  \footnotesize

  \begin{itemize}
    \item \textbf{XGB-only} --- modeling moved from ridge+XGB embed to a single \textbf{XGBoost} model for both optimization and evaluation.
    \item \textbf{Level objective} --- MILP maximizes $\sum_s f_s(x_s)$, not incremental lift $f(x)-f(0)$. Incremental penalizes segments with large intercepts; levels align predicted conversions more closely with actuals.
    \item \textbf{Observed-budget prediction floors} --- predict zero when below each segment's historical minimum \texttt{daily\_budget}.
  \end{itemize}
  \normalsize
\end{frame}

\begin{frame}{Two-stage backtest architecture}
    \textbf{Goal:} keyword sets tend to stay mostly stable over the backtest window, so we can optimize them once and then re-optimize budgets daily. Also aligns with current practices.
  \begin{itemize}
    \item \textbf{Stage~1:} optimize one keyword set per segment over the full backtest window via a multi-day \textbf{XGB tree-embed} MILP with per-day budgets.
    \item \textbf{Stage~2:} fix stage-1 keyword sets and re-optimize budgets daily with walk-forward refit.
  \end{itemize}
\end{frame}

\begin{frame}{Keyword set pipeline}
  \textbf{Goal:} richer candidate generation and tighter filtering so model doesn't extrapolate too much.

  \begin{itemize}
    \item \textbf{Enrollment allowlist} gates which keywords enter the candidate pool based on historically successful keywords (~50 keywords).
    \item Replaced click-ranked synthetics with \textbf{top-conversion} and \textbf{allowlist-based} synthetic sets.
    \item \textbf{Per-match-type synthetics} generated from the region panel with verification.
  \end{itemize}
\end{frame}

\begin{frame}{Modeling, CV \& features}
  \footnotesize
  \textbf{Goal:} tighten how models are trained, validated, and wired into the optimizer.

  \begin{itemize}
    \item \textbf{Expanding-window time-series CV} on the training calendar (3 folds, $\geq$50\% train fraction); fold date logging retained.
    \item Training panel filtered to allowed optimization segments; \texttt{is\_broad\_match} replaces multi-hot match flags.
    \item Ablations and tighter CV tuning grids.
    \item Optional recency half-life sample weights.
  \end{itemize}
  \normalsize
\end{frame}

\begin{frame}{Model $R^2$ results (XGB, tuned)}
  \footnotesize
  \texttt{conv\_scaled\_clicks}; deduped 10-feat; 75-day holdout; 3-fold expanding-window CV.

  \vspace{0.5em}
  \begin{tabular}{@{}lrr@{}}
    \toprule
    Panel / change & CV $R^2$ & Ho $R^2$ \\
    \midrule
    Full panel (12 segments, incl.\ out-of-scope) & $-$0.09 & 0.79 \\
    Optimization scope (6 segments), old grid & 0.41 & 0.15 \\
    Optimization scope, \textbf{current grid} & \textbf{0.42} & \textbf{0.25} \\
    \bottomrule
  \end{tabular}

  \vspace{0.5em}
  \begin{itemize}
    \item \textbf{Scope filter} --- drop region~C and non-optimized match-type configs; training panel matches the 6-segment MILP. Full panel gave \textbf{negative CV $R^2$} with \textbf{misleading positive holdout}.
    \item \textbf{Grid tightening} --- old grid (\texttt{max\_depth=4}) inflated CV ($\sim$0.41) while holdout collapsed ($\sim$0.15); current grid (\texttt{max\_depth} $\in\{2,3\}$, \texttt{n\_estimators} $\leq 20$) is more balanced.
  \end{itemize}
  \normalsize
\end{frame}
