\begin{frame}{Optimization formulation}
  \small
  \textbf{Segments} $s \in \mathcal{S}$: region $\times$ match type (6 active: USA, A, B $\times$ Broad, Phrase; Exact).
  \textbf{Candidates} $\mathcal{K}_s$: historical + synthetic keyword sets for $s$ (see next slide).

  \textbf{Decision variables:}
  $x_s \geq 0$ daily budget;\;
  $y_{s,k} \in \{0,1\}$ with $\sum_{k \in \mathcal{K}_s} y_{s,k} = 1$.

  \textbf{Ungated level} (\texttt{ensemble\_ridge\_xgb}, \texttt{tree\_embed}; target \texttt{conv\_scaled\_clicks}):
  $\tilde{f}_{s,k}(x_s)=\hat f_{\mathrm{opt}}(x_s,\mathbf{z}^{\mathrm{cal}},\boldsymbol{\phi}_k)$.

  \textbf{Observed-budget floor} (\texttt{apply\_observed\_budget\_floor}${=}$\texttt{true}):
  let $x_s^{\min}$ = min historical \texttt{daily\_budget} for segment $s$ in the training panel.\\
  Embedded MILP level (and post-solve check):
  \[
    \widehat{F}_s(x_s)=
    \begin{cases}
      0 & \text{if } x_s < x_s^{\min} \\
      \tilde{f}_s(x_s) & \text{otherwise}
    \end{cases}
  \]
  (McCormick gating in Gurobi; same rule for evaluation $\hat f$ in \texttt{plan\_vs\_actual}.)

  \textbf{MILP objective} (\texttt{evaluation.objective}${=}$\texttt{levels}):
  $\max \sum_s \widehat{F}_s(x_s) - \lambda\sum_s x_s$.
  \textbf{Plan scoring} (evaluation ensemble, not the MILP): code stores
  $\hat f(x_s,\mathbf{z})-\hat f(0,\mathbf{z})$ at \texttt{baseline\_budget}${=}0$; with the observed-budget floor, $\hat f(0){=}0$ $\Rightarrow$ headline totals $\approx$ gated \textbf{levels} (not a separate incremental objective).

  \textbf{Constraints:}
  $\sum_s x_s \leq B$;\;
  $x^{\min}_s \leq x_s \leq x^{\max}_s$;\;
  USA $\geq$ A $\geq$ B spend;\;
  $\sum_k y_{s,k}=1$.
  {\scriptsize $\lambda{=}10^{-4}$ tiebreak; May two-stage stage~2 uses fixed course cap ${\sim}\$413$/day (\texttt{--budget}); panel caps ${\sim}\$367$ for actual rows.}
  \normalsize
\end{frame}

\begin{frame}{Backtest: data and decisions}
  \textbf{History used}
  \begin{itemize}
    \item Segment-day panel: configured \textbf{daily budget cap}, keyword list in use, clicks,
      scaled clicks, and related fields.
    \item Per-segment \textbf{candidate keyword lists} (historical + allowlist-filtered
      \texttt{top\_conv} / allowlist / semantic / dispersion / composite at $N\in\{10,20,40\}$).
    \item Per-list \textbf{context}: calendar for the planning day, search-volume / competition
      aggregates, embedding cohesion and course similarity.
  \end{itemize}

  \vspace{0.35em}
  \textbf{Decisions}
  \begin{itemize}
    \item \textbf{Stage 1} (once): binary $y_{s,k}$ --- one keyword list per segment for the window.
    \item \textbf{Stage 2} (each day $t$): continuous $x_s$ --- daily budget per segment ($y_{s,k}$ fixed).
  \end{itemize}

  \vspace{0.35em}
  \textbf{Constraints:} $\sum_s x_s \leq B$; regional order USA $\geq$ A $\geq$ B;
  $x_s^{\min}\leq x_s\leq x_s^{\max}$; predict $0$ if $x_s$ below min observed cap (floor).
\end{frame}

\begin{frame}{Backtest: two models (do not mix)}
  \textbf{1.\ Optimizer model} --- drives the MILP only
  \begin{itemize}
    \item Form: \textbf{ridge + XGBoost blend}, embedded exactly in Gurobi (nonlinear trees +
      linear ridge part).
    \item \textbf{Stage 1:} fit once on \texttt{date}${<}t_0$; \textbf{stage 2:} refit daily on \texttt{date}${<}t$ (budget MILP only).
    \item Hyperparameters: \textbf{3-fold expanding time-series CV} on that training slice
      (separate grids for ridge and XGB).
    \item Features at planning day $t$: calendar, list-level GKP/embedding stats, budget.
  \end{itemize}

  \vspace{0.35em}
  \textbf{2.\ Evaluation model} --- scores plans only; \textbf{never} chooses budgets/sets
  \begin{itemize}
    \item Form: \textbf{five-model ensemble} (ridge, power-log, power-level, random forest,
      XGBoost).
    \item \textbf{Fit once} on the \textbf{full modeling panel} (2{,}258 segment-days,
      2024-06-26 $\rightarrow$ 2026-05-25), including the May backtest window itself.
    \item Weights: inverse tournament CV-RMSE over the five base members (fixed for the window).
    \item Same $\hat f$ for every May day: a stable scoring ruler (optimizer still walk-forward).
  \end{itemize}
\end{frame}

\begin{frame}{Models}
  \footnotesize
  \textbf{Target:} \texttt{conv\_scaled\_clicks}.
  \textbf{Context} $\mathbf{z}$: planning-day calendar + keyword-set features (GKP, embeddings).

  \vspace{0.2em}
  \textbf{Evaluation ensemble} (backtest / \texttt{plan\_vs\_actual}; fit once on full panel):
  \begin{align*}
    \hat f(x_s,\mathbf{z})
      &= \textstyle\sum_m w_m \hat f_m(x_s,\mathbf{z}) \big/ \textstyle\sum_m w_m,
      \quad w_m \propto 1/\mathrm{CV\mbox{-}RMSE}_m,\\
    \mathcal{M} &= \{\text{ridge},\,\text{power\_log},\,\text{power\_level},\,\text{RF},\,\text{XGB}\}.
  \end{align*}
  \textbf{Backtest columns} (\texttt{plan\_vs\_actual}):
  \texttt{pred\_lift\_raw}$=\hat f(\text{plan})-\hat f(0)$,
  \texttt{f\_plan\_level}$=\hat f(\text{plan})$ (gated).
  With \texttt{baseline\_budget}${=}0$ and the floor, $\hat f(0){=}0$ so the two coincide; clipped \texttt{pred\_lift}${=}\max(\cdot,0)$.

  \vspace{0.15em}
  \begin{itemize}
    \item \textbf{ridge} (MILP-linear design):
      $y=\alpha_s+\beta_s x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$ on a
      \textbf{scaled} design matrix ($x_s$ and numeric $\mathbf{z}$ standardized at fit);
      Gurobi uses per-candidate affine levels from \texttt{predict\_design\_frame}.
    \item \textbf{power\_log}: $\log(1{+}y)=\alpha_s+\beta_s\log x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$;
      predict $\hat y=\mathrm{expm1}(\cdot)$ (elasticity in log--log space).
    \item \textbf{power\_level}: $y=\alpha_s+\beta_s\log x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$
      (Ridge on levels; budget feature $=\log x_s$).
    \item \textbf{random\_forest / xgboost}: $\hat y = g(\text{region},\,\text{match},\,x_s,\,\mathbf{z})$
      (trees on scaled $x_s$, OHE region, numeric/categorical $\mathbf{z}$).
  \end{itemize}
  {\scriptsize Optimizer MILP: \texttt{ensemble\_ridge\_xgb} (\texttt{tree\_embed}); 5-member ensemble is scoring-only for \texttt{plan\_vs\_actual}.}

  \vspace{0.1em}
  \normalsize
\end{frame}

\begin{frame}{Caveats}
  \begin{itemize}
    \item \textbf{$\hat f$ $\neq$ observed:} plan gated $\hat f$ $\sim$24.5 / panel $\hat f$ $\sim$2.5 vs.\ realized \texttt{conv\_scaled} $\sim$2.6; plan level is far above realized in this window.
    \item \textbf{Scoring vs.\ optimizer mismatch:} MILP uses \texttt{ensemble\_ridge\_xgb}; evaluation uses the 5-member ensemble --- plan $\hat f$ $\gg$ panel $\hat f$ ($+876\%$).
    \item \textbf{Fixed synthetic sets:} stage~1 picks \texttt{composite\_n40} (Broad) and \texttt{allowlist\_n40} (Phrase; Exact) per region --- unchanged all 14 days.
    \item \textbf{Budget mode:} two-stage plan uses fixed course cap ${\sim}\$413$/day; panel configured caps ${\sim}\$367$/day for actual rows.
  \end{itemize}
\end{frame}
