\begin{frame}{Optimization formulation}
  \small
  \textbf{Segments} $s \in \mathcal{S}$: region $\times$ match type (6 active: USA, A, B $\times$ Broad, Phrase; Exact).
  \textbf{Candidates} $\mathcal{K}_s$: historical + synthetic keyword sets for $s$ (see next slide).

  \textbf{Decision variables:}
  $x_s \geq 0$ daily budget;\;
  $y_{s,k} \in \{0,1\}$ with $\sum_{k \in \mathcal{K}_s} y_{s,k} = 1$.

  \textbf{Response} (\texttt{ensemble\_ridge\_xgb}, \texttt{tree\_embed}; target \texttt{conv\_scaled\_clicks}; planning-day calendar fixed):
  \[
    \widehat{y}_{s,k}(x_s)
      = \hat f_{\mathrm{opt}}\!\bigl(x_s,\, \mathbf{z}^{\mathrm{cal}},\, \boldsymbol{\phi}_k\bigr),
    \qquad
    \boldsymbol{\phi}_k \text{: set-level semantic + GKP features.}
  \]

  \textbf{Incremental lift} at baseline budget $b_0$ (\texttt{evaluation.baseline\_budget}${=}0$):
  $\Delta_{s,k}(x_s)=\widehat{F}_{s,k}(x_s)-\widehat{F}_{s,k}(b_0)$ for the chosen list $\boldsymbol{\phi}_k$.

  \textbf{Problem (MILP):}
  \[
    \max_{\{x_s,\,y_{s,k}\}} \sum_{s,k} y_{s,k}\,\Delta_{s,k}(x_s)
      - \lambda \sum_s x_s
    \quad \text{s.t.}\quad
    \sum_s x_s \leq B,\;
    x^{\min}_s \leq x_s \leq x^{\max}_s,
  \]
  \[
    \sum_{s:\, r(s)=\mathrm{USA}} \!\!x_s
    \geq
    \sum_{s:\, r(s)=A} \!\!x_s
    \geq
    \sum_{s:\, r(s)=B} \!\!x_s,
    \qquad
    \sum_{k \in \mathcal{K}_s} y_{s,k} = 1 \;\;\forall s.
  \]
  {\scriptsize $\lambda{=}10^{-4}$ budget tiebreak; $B{=}412.7$/day (configured cap, constant Mar 2026).}
  \normalsize
\end{frame}

\begin{frame}{Keyword set candidates}
  \textbf{Pool $\mathcal{K}_s$} (\texttt{build\_keyword\_candidates.py}):
  \begin{itemize}
    \item \textbf{Historical} --- every \texttt{keyword\_set\_id} seen in segment (change history + panel).
    \item \textbf{synthetic\_top} --- top-click / top-efficiency keywords from \texttt{kw-day-panel}, filtered to region and allowed match types.
    \item \textbf{synthetic\_semantic} --- top keywords by course-anchor similarity; size $\approx$ median historical list length.
    \item \textbf{synthetic\_dispersion} --- greedy subset maximizing embedding spread.
    \item \textbf{synthetic\_composite} --- greedy set maximizing normalized course similarity + dispersion.
  \end{itemize}

  \vspace{0.35em}
  \textbf{Lists \& features:} match-type columns (\texttt{broad}/\texttt{phrase}/\texttt{exact} keywords assigned by dominant clicks); GKP aggregates + embedding cohesion/dispersion/course similarity merged in \texttt{build\_gkp\_set\_features.py} $\rightarrow$ \texttt{keyword-set-features.csv}.

  \vspace{0.25em}
  {\footnotesize MILP picks \textbf{one set per segment per day}; choice can change when the walk-forward optimizer refit shifts $\hat f$ (see \textbf{Keyword sets in March plans}).}
\end{frame}

\begin{frame}{Backtest: scope}
  \textbf{What we simulate:} daily campaign re-optimization over March 2026 (31 days).\\
  \textbf{Course / experiment:} System Thinking ads; default parameter set.

  \vspace{0.4em}
  \textbf{Outcome modeled:} \emph{conversion-scaled clicks} --- clicks multiplied by each
  segment's historical conversion-per-click (more stable than raw conversions).

  \vspace{0.4em}
  \textbf{Budget:} \$412.70 total configured daily cap (sum of segment caps from change history;
  constant all month).

  \vspace{0.4em}
  \textbf{Segments (6):} region $\times$ match type --- USA, A, B each split into Broad vs.\
  Phrase; Exact (region C excluded).
\end{frame}

\begin{frame}{Backtest: data and decisions}
  \textbf{History used}
  \begin{itemize}
    \item Segment-day panel: configured \textbf{daily budget cap}, keyword list in use, clicks,
      scaled clicks, and related fields.
    \item Per-segment \textbf{candidate keyword lists} (every list ever used plus synthetic
      pools: top performers, semantic, dispersion, composite).
    \item Per-list \textbf{context}: calendar for the planning day, search-volume / competition
      aggregates, embedding cohesion and course similarity.
  \end{itemize}

  \vspace{0.35em}
  \textbf{Decisions each day} (one MILP solve per calendar day $t$)
  \begin{itemize}
    \item Continuous: daily budget $x_s \geq 0$ per segment (bounded by historical min--max caps).
    \item Binary: exactly one keyword list $k$ per segment.
  \end{itemize}

  \vspace{0.35em}
  \textbf{Constraints:} $\sum_s x_s \leq B$; regional caps USA $\geq$ A $\geq$ B;
  tiny budget penalty breaks ties when predicted response is flat.
\end{frame}

\begin{frame}{Backtest: two models (do not mix)}
  \textbf{1.\ Optimizer model} --- drives the MILP only
  \begin{itemize}
    \item Form: \textbf{ridge + XGBoost blend}, embedded exactly in Gurobi (nonlinear trees +
      linear ridge part).
    \item \textbf{Refit every day} on all segment-days with date $< t$.
    \item Hyperparameters: \textbf{3-fold expanding time-series CV} on that training slice
      (separate grids for ridge and XGB).
    \item Features at planning day $t$: calendar, list-level GKP/embedding stats, budget.
  \end{itemize}

  \vspace{0.35em}
  \textbf{2.\ Evaluation model} --- scores plans only; \textbf{never} chooses budgets/sets
  \begin{itemize}
    \item Form: \textbf{five-model ensemble} (ridge, power-log, power-level, random forest,
      XGBoost).
    \item \textbf{Fit once} on the \textbf{full modeling panel} (2{,}258 segment-days,
      2024-06-26 $\rightarrow$ 2026-05-25), including dates \textbf{after} March 2026.
    \item Weights: inverse tournament CV-RMSE over the five base members (fixed for the month).
    \item Same $\hat f$ for every March day: a stable scoring ruler (optimizer still walk-forward).
  \end{itemize}
\end{frame}

\begin{frame}{Backtest: daily procedure}
  \footnotesize
  \textbf{Before day 1:}
  \begin{enumerate}
    \item Offline tournament (separate step) picks hyperparameter grids and records CV metrics
      for all candidate model families.
    \item Fit the evaluation ensemble on the full panel (through May 2026); fix weights.
  \end{enumerate}

  \vspace{0.25em}
  \textbf{For each day $t \in$ Mar 2026} (abort if fewer than 50 training rows before $t$):
  \begin{enumerate}
    \item \textbf{Optimize:} train on date $< t$ $\rightarrow$ CV-tune $\rightarrow$ fit ridge+XGB
      blend $\rightarrow$ solve MILP $\rightarrow$ plan (budgets + keyword lists).
    \item \textbf{Score:} with the \textbf{fixed} evaluation ensemble, compute
      \textbf{plan lift} $\hat f(\text{plan budget},\text{plan set}) - \hat f(0,\text{plan set})$ and
      \textbf{actual lift} $\hat f(\text{panel budget},\text{panel set}) - \hat f(0,\text{panel set})$.
    \item \textbf{Observe:} sum realized scaled clicks from the panel on day $t$.
  \end{enumerate}

  \vspace{0.25em}
  \textbf{Month rollup:} mean daily plan vs.\ actual model lift (raw totals), budget shares by region,
  keyword-list choices, RMSE of predicted lift vs.\ observed levels.
  \normalsize
\end{frame}

\begin{frame}{Models}
  \footnotesize
  \textbf{Target:} \texttt{conv\_scaled\_clicks}.
  \textbf{Context} $\mathbf{z}$: planning-day calendar + keyword-set features (GKP, embeddings).

  \vspace{0.2em}
  \textbf{Evaluation ensemble} (backtest / \texttt{plan\_vs\_actual}; fit once on full panel):
  \begin{align*}
    \hat f(x_s,\mathbf{z})
      &= \textstyle\sum_m w_m \hat f_m(x_s,\mathbf{z}) \big/ \textstyle\sum_m w_m,
      \quad w_m \propto 1/\mathrm{CV\mbox{-}RMSE}_m,\\
    \mathcal{M} &= \{\text{ridge},\,\text{power\_log},\,\text{power\_level},\,\text{RF},\,\text{XGB}\}.
  \end{align*}
  \textbf{Lift:}
  $\mathrm{lift}=\max\bigl(\hat f(x_s^{\mathrm{dec}},\mathbf{z})-\hat f(0,\mathbf{z}),\,0\bigr)$
  (counterfactual at \$0 budget with the \textbf{same} keyword-set features $\mathbf{z}$:
  plan set for plan, panel set for actual).

  \vspace{0.15em}
  \begin{itemize}
    \item \textbf{ridge} (MILP-linear design):
      $y=\alpha_s+\beta_s x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$ on a
      \textbf{scaled} design matrix ($x_s$ and numeric $\mathbf{z}$ standardized at fit);
      Gurobi uses per-candidate affine levels from \texttt{predict\_design\_frame}.
    \item \textbf{power\_log}: $\log(1{+}y)=\alpha_s+\beta_s\log x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$;
      predict $\hat y=\mathrm{expm1}(\cdot)$ (elasticity in log--log space).
    \item \textbf{power\_level}: $y=\alpha_s+\beta_s\log x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$
      (Ridge on levels; budget feature $=\log x_s$).
    \item \textbf{random\_forest / xgboost}: $\hat y = g(\text{region},\,\text{match},\,x_s,\,\mathbf{z})$
      (trees on scaled $x_s$, OHE region, numeric/categorical $\mathbf{z}$).
  \end{itemize}
  {\scriptsize Optimizer MILP: \texttt{ensemble\_ridge\_xgb} (\texttt{tree\_embed}); 5-member ensemble is scoring-only for \texttt{plan\_vs\_actual}.}

  \vspace{0.1em}
  \normalsize
\end{frame}

\begin{frame}{What we built}
  \begin{enumerate}
    \item \textbf{Panel} --- \texttt{campaign-day-panel}: version-linked budgets, clicks, \texttt{all\_conv} per segment.
    \item \textbf{Features} --- calendar; keyword-set embeddings; GKP aggregates.
    \item \textbf{Response tournament} --- ridge, power (log/level), RF, XGB; time-series CV; level-scale RMSE.
    \item \textbf{MILP} --- per-segment budget + keyword set; maximize incremental lift at $b_0{=}0$; Gurobi \texttt{tree\_embed}.
    \item \textbf{Backtest} --- daily \texttt{plan\_vs\_actual.csv} (ensemble scoring) $\rightarrow$ \texttt{backtest\_summary.csv}.
  \end{enumerate}
\end{frame}

\begin{frame}{Modeling decisions}
  \begin{itemize}
    \item \textbf{Target:} \texttt{conv\_scaled\_clicks} not raw \texttt{all\_conv} --- conversions are regime-heavy; clicks more stable; scaling weights toward segment conv efficiency.
    \item \textbf{Lever:} \texttt{daily\_budget} (configured cap), not cost.
    \item \textbf{Validation:} time-series CV (3 folds, $\geq$50\% train fraction).
    \item \textbf{Production path:} \texttt{optimizer\_winner: ensemble\_ridge\_xgb}; evaluation 5-member ensemble with CV-RMSE weights.
  \end{itemize}
\end{frame}

\begin{frame}{Caveats}
  \begin{itemize}
    \item \textbf{Lift $\neq$ observed level:} predicted daily lifts $\sim$3.9 (plan) / $\sim$1.4 (panel config under $\hat f$) vs.\ observed $\sim$15; scoring is counterfactual on $\hat f$, not realized outcomes.
    \item \textbf{Scoring vs.\ optimizer mismatch:} MILP uses \texttt{ensemble\_ridge\_xgb}; evaluation uses the wider 5-member ensemble --- plan lift can exceed actual lift under $\hat f$.
    \item \textbf{Keyword sets not fixed:} walk-forward refit shifts choices (e.g.\ A Broad splits across historical \texttt{ks\_0001} and \texttt{semantic\_2}); USA Phrase+Exact mixes \texttt{ks\_0010} with dispersion/semantic synthetics.
    \item \textbf{Budget binding:} plan totals hit the \$413/day cap; ${\sim}$0.5 / 6 segments at \$0 budget on average (others absorb spend).
  \end{itemize}
\end{frame}

\begin{frame}{Next steps}
  \begin{enumerate}

    \item \textbf{Two-stage backtest:} fixed sets for period + weekly budgets.
  \end{enumerate}
\end{frame}
