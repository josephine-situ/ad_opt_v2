\begin{frame}{Optimization formulation}
  \small
  \textbf{Segments} $s \in \mathcal{S}$: region $\times$ match type (6 active: USA, A, B $\times$ Broad, Phrase; Exact).
  \textbf{Candidates} $\mathcal{K}_s$: historical + synthetic keyword sets for $s$ (see next slide).

  \textbf{Decision variables:}
  $x_s \geq 0$ daily budget;\;
  $y_{s,k} \in \{0,1\}$ with $\sum_{k \in \mathcal{K}_s} y_{s,k} = 1$.

  \textbf{Response} (\texttt{ensemble\_ridge\_xgb}, \texttt{tree\_embed}; target \texttt{conv\_scaled\_clicks}; planning-day calendar fixed):
  \[
    \widehat{y}_{s,k}(x_s)
      = \hat f_{\mathrm{opt}}\!\bigl(x_s,\, \mathbf{z}^{\mathrm{cal}},\, \boldsymbol{\phi}_k\bigr),
    \qquad
    \boldsymbol{\phi}_k \text{: set-level semantic + GKP features.}
  \]

  \textbf{Problem (MILP):}
  \[
    \max_{\{x_s,\,y_{s,k}\}} \sum_{s,k} y_{s,k}\,\widehat{y}_{s,k}(x_s)
      - \lambda \sum_s x_s
    \quad \text{s.t.}\quad
    \sum_s x_s \leq B,\;
    x^{\min}_s \leq x_s \leq x^{\max}_s,
  \]
  \[
    \sum_{s:\, r(s)=\mathrm{USA}} \!\!x_s
    \geq
    \sum_{s:\, r(s)=A} \!\!x_s
    \geq
    \sum_{s:\, r(s)=B} \!\!x_s,
    \qquad
    \sum_{k \in \mathcal{K}_s} y_{s,k} = 1 \;\;\forall s.
  \]
  {\scriptsize $\lambda{=}10^{-4}$ budget tiebreak; $B{=}412.7$/day (configured cap, constant Mar 2026).}
  \normalsize
\end{frame}

\begin{frame}{Keyword set candidates}
  \textbf{Pool $\mathcal{K}_s$} (\texttt{build\_keyword\_candidates.py}):
  \begin{itemize}
    \item \textbf{Historical} --- every \texttt{keyword\_set\_id} seen in segment (change history + panel).
    \item \textbf{synthetic\_top} --- top-click / top-efficiency keywords from \texttt{kw-day-panel}, filtered to region and allowed match types.
    \item \textbf{synthetic\_semantic} --- top keywords by course-anchor similarity; size $\approx$ median historical list length.
    \item \textbf{synthetic\_dispersion} --- greedy subset maximizing embedding spread.
    \item \textbf{synthetic\_composite} --- union of top + semantic + dispersion pools.
  \end{itemize}

  \vspace{0.35em}
  \textbf{Lists \& features:} match-type columns (\texttt{broad}/\texttt{phrase}/\texttt{exact} keywords assigned by dominant clicks); GKP aggregates + embedding cohesion/dispersion/course similarity merged in \texttt{build\_gkp\_set\_features.py} $\rightarrow$ \texttt{keyword-set-features.csv}.

  \vspace{0.25em}
  {\footnotesize MILP picks \textbf{one set per segment} per day; \texttt{ensemble\_ridge\_xgb} fit on full panel, plans solved walk-forward (\texttt{date}${<}t$ train rows for features).}
\end{frame}

\begin{frame}{Backtest at a glance}
  \textbf{Window:} 2026-03-01 -- 2026-03-31 (31 days), \texttt{sys\_think} / \texttt{default}.\\
  \textbf{Budget cap:} \$412.70/day (sum of segment \texttt{daily\_budget} from change history; constant all 31 days).\\
  \textbf{objective:} \texttt{conv\_scaled\_clicks} (clicks $\times$ historical conv/click by segment).

  \vspace{0.5em}
  \textbf{Daily walk-forward:} for each $t$, train on \texttt{date} $< t$, solve MILP (budget + keyword set), score plan vs.\ panel actuals.

  \vspace{0.5em}
  \textbf{Evaluation:} 5-member CV-weighted \textbf{ensemble} (see \textbf{Models}); fit once on full panel. Optimizer: \texttt{ensemble\_ridge\_xgb} (\texttt{tree\_embed}), reused daily.
\end{frame}

\begin{frame}{Models}
  \footnotesize
  \textbf{Target:} \texttt{conv\_scaled\_clicks}.
  \textbf{Context} $\mathbf{z}$: planning-day calendar + keyword-set features (GKP, embeddings).

  \vspace{0.2em}
  \textbf{Evaluation ensemble} (backtest / \texttt{plan\_vs\_actual}; fit once on full panel):
  \begin{align*}
    \hat f(x_s,\mathbf{z})
      &= \textstyle\sum_m w_m \hat f_m(x_s,\mathbf{z}) \big/ \textstyle\sum_m w_m,
      \quad w_m \propto 1/\mathrm{CV\mbox{-}RMSE}_m,\\
    \mathcal{M} &= \{\text{ridge},\,\text{power\_log},\,\text{power\_level},\,\text{RF},\,\text{XGB}\}.
  \end{align*}
  \textbf{Lift:}
  $\mathrm{lift}=\max\bigl(\hat f(x_s^{\mathrm{dec}},\mathbf{z})-\hat f(0,\mathbf{z}),\,0\bigr)$
  (counterfactual at \$0 budget; same $\mathbf{z}$ for plan vs.\ actual).

  \vspace{0.15em}
  \begin{itemize}
    \item \textbf{ridge} (MILP-linear design):
      $y=\alpha_s+\beta_s x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$;
      $\alpha_s,\beta_s$ from region + match flags + $x_s\times$(region, match) + $\mathbf{z}$.
    \item \textbf{power\_log}: $\log(1{+}y)=\alpha_s+\beta_s\log x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$;
      predict $\hat y=\mathrm{expm1}(\cdot)$ (elasticity in log--log space).
    \item \textbf{power\_level}: $y=\alpha_s+\beta_s\log x_s+\mathbf{z}^\top\boldsymbol{\gamma}+\varepsilon$
      (Ridge on levels; budget feature $=\log x_s$).
    \item \textbf{random\_forest / xgboost}: $\hat y = g(\text{region},\,\text{match},\,x_s,\,\mathbf{z})$
      (trees on scaled $x_s$, OHE region, numeric/categorical $\mathbf{z}$).
  \end{itemize}
  {\scriptsize Optimizer MILP: \texttt{ensemble\_ridge\_xgb} (\texttt{tree\_embed}); 5-member ensemble is scoring-only for \texttt{plan\_vs\_actual}.}

  \vspace{0.1em}
  \normalsize
\end{frame}

\begin{frame}{What we built}
  \begin{enumerate}
    \item \textbf{Panel} --- \texttt{campaign-day-panel}: version-linked budgets, clicks, \texttt{all\_conv} per segment.
    \item \textbf{Features} --- calendar; keyword-set embeddings; GKP aggregates.
    \item \textbf{Response tournament} --- ridge, power (log/level), RF, XGB; time-series CV; level-scale RMSE.
    \item \textbf{MILP} --- per-segment budget + keyword set; Gurobi \texttt{tree\_embed}.
    \item \textbf{Backtest} --- daily \texttt{plan\_vs\_actual.csv} (ensemble scoring) $\rightarrow$ \texttt{backtest\_summary.csv}.
  \end{enumerate}
\end{frame}

\begin{frame}{Modeling decisions}
  \begin{itemize}
    \item \textbf{Target:} \texttt{conv\_scaled\_clicks} not raw \texttt{all\_conv} --- conversions are regime-heavy; clicks more stable; scaling weights toward segment conv efficiency.
    \item \textbf{Lever:} \texttt{daily\_budget} (configured cap), not cost.
    \item \textbf{Validation:} time-series CV (3 folds, $\geq$50\% train fraction).
    \item \textbf{Production path:} \texttt{optimizer\_winner: ensemble\_ridge\_xgb}; evaluation 5-member ensemble with CV-RMSE weights.
  \end{itemize}
\end{frame}

\begin{frame}{Caveats}
  \begin{itemize}
    \item \textbf{Lift $\neq$ observed level:} predicted daily lifts $\mathcal{O}(2)$--$6$ vs.\ observed $\sim$15; scoring is counterfactual on $\hat f$, not realized outcomes.
    \item \textbf{Scoring vs.\ optimizer mismatch:} MILP uses \texttt{ensemble\_ridge\_xgb}; evaluation uses the wider 5-member ensemble --- plan lift can exceed actual lift under $\hat f$.
    \item \textbf{Set stability:} fixed synthetic keyword set per segment all 31 days (dispersion / semantic pools).
    \item \textbf{Budget binding:} all segments receive positive budget; plan totals hit the \$413/day cap.
  \end{itemize}
\end{frame}

\begin{frame}{Next steps}
  \begin{enumerate}

    \item \textbf{Two-stage backtest:} fixed sets for period + weekly budgets.
  \end{enumerate}
\end{frame}
